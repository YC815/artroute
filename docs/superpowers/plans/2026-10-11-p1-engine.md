# P1-engine：分治吸附、串接、評分與 `/match` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 引擎提供 `POST /match`：輸入經緯度筆畫，回傳一條連續、合法的自行車路線（含分段種類與貼合度），並在台灣真實路網上排除快速道路、用基準測試集量化品質。

**Architecture:** 分治吸附（整段 trace → 失敗對半切 → 最小段合法繞路）產生每個部件的路段鏈；串接器用直線距離估計排出順序與封閉圈進出點，再以 Valhalla `route` 補連接線；評分在局部平面上算覆蓋與乾淨。每個 uvicorn worker 依 `/data/current` 快取自己的 `Router`。建圖前用 pyosmium 把快速道路標成 `bicycle=no`。

**Tech Stack:** Python 3.13、uv、FastAPI、pydantic v2、pyvalhalla 3.9.1、numpy、osmium（pyosmium 4.x）、pytest。

**對應設計文件：** `docs/superpowers/specs/2026-10-10-strava-art-design.md` §3（併發模型）、§4.3、§4.4、§4.5、§4.7、§5、§6.1、§6.3、§8。

## Global Constraints

- Python 一律用 uv：`uv add`、`uv run`；不用 pip、不用系統 python。Python 3.13；pyvalhalla 釘 `3.9.1`。

```python
"""每個 worker 一份：依 /data/current 提供 Router，換版時自動換新（也就是熱重載）。"""

import threading
from collections.abc import Callable
from pathlib import Path

from engine.data.layout import DataLayout
from engine.routing.router import Router


class NoDataError(Exception):
    pass


class CurrentRouter[R]:
    def __init__(self, layout: DataLayout, factory: Callable[[Path], R] = Router.from_build) -> None:
        self._layout = layout
        self._factory = factory
        self._lock = threading.Lock()
        self._build: Path | None = None
        self._router: R | None = None

    def get(self) -> tuple[R, str]:
        build = self._layout.current()
        if build is None or not build.is_dir():
            raise NoDataError("路網資料尚未就緒")
        with self._lock:
            if build != self._build or self._router is None:
                # 先建好新的再替換：正在用舊 Router 的請求會在舊實例上跑完
                self._router = self._factory(build)
                self._build = build
            return self._router, build.name
```

```python
from collections.abc import Callable


class MatchTimeout(Exception):
    pass


class Deadline:
    def __init__(self, seconds: float, clock: Callable[[], float] = time.monotonic) -> None:
        self._clock = clock
        self._end = clock() + seconds
```

```python
"""單一部件的分治吸附（spec §4.3）。

整段 trace → 失敗就對半切 → 短於 min_piece_m 仍失敗的小段改用合法 route 繞路。
相鄰段落接不上時用 route 補；補不起來就斷成新的 chain，絕不畫直線。
"""

import math
from dataclasses import dataclass

from engine.matching.deadline import Deadline
from engine.matching.types import JOIN_TOLERANCE_M, MatchRouter, MatchWarning, Segment, segment_from
from engine.routing.geo import haversine_m, polyline_length_m
from engine.routing.router import LonLat, RoutingError

DENSIFY_STEP_M = 25.0


@dataclass(frozen=True)
class StrokeMatch:
    chains: list[list[Segment]]
    warnings: list[MatchWarning]


@dataclass(frozen=True)
class _Context:
    router: MatchRouter
    stroke_index: int
    search_radius_m: float
    min_piece_m: float
    deadline: Deadline


def densify(points: list[LonLat], step_m: float = DENSIFY_STEP_M) -> list[LonLat]:
    out = [points[0]]
    for a, b in zip(points, points[1:]):
        n = max(1, math.ceil(haversine_m(a, b) / step_m))
        out.extend((a[0] + (b[0] - a[0]) * i / n, a[1] + (b[1] - a[1]) * i / n) for i in range(1, n + 1))
    return out


def _route(ctx: _Context, start: LonLat, end: LonLat) -> Segment | None:
    ctx.deadline.check()
    try:
        return segment_from("detour", ctx.stroke_index, ctx.router.route(start, end))
    except RoutingError:
        return None


def _solve(ctx: _Context, piece: list[LonLat]) -> list[Segment | None]:
    """回傳依序排列的段落；None 代表這一小段找不到任何合法路徑。"""
    ctx.deadline.check()
    try:
        return [segment_from("stroke", ctx.stroke_index, ctx.router.trace(piece, search_radius_m=ctx.search_radius_m))]
    except RoutingError:
        pass
    if len(piece) < 3 or polyline_length_m(piece) <= ctx.min_piece_m:
        return [_route(ctx, piece[0], piece[-1])]
    mid = len(piece) // 2
    return _solve(ctx, piece[: mid + 1]) + _solve(ctx, piece[mid:])


def _link(ctx: _Context, pieces: list[Segment | None]) -> list[list[Segment]]:
    chains: list[list[Segment]] = []
    chain: list[Segment] = []
    for segment in pieces:
        if segment is None:
            continue
        if chain and haversine_m(chain[-1].points[-1], segment.points[0]) > JOIN_TOLERANCE_M:
            bridge = _route(ctx, chain[-1].points[-1], segment.points[0])
            if bridge is None:
                chains.append(chain)
                chain = []
            else:
                chain.append(bridge)
        chain.append(segment)
    if chain:
        chains.append(chain)
    return chains


def match_stroke(
    router: MatchRouter,
    points: list[LonLat],
    *,
    stroke_index: int,
    search_radius_m: float,
    min_piece_m: float,
    deadline: Deadline,
) -> StrokeMatch:
    ctx = _Context(router, stroke_index, search_radius_m, min_piece_m, deadline)
    pieces = _solve(ctx, densify(points))
    chains = _link(ctx, pieces)
    if not chains:
        warning = MatchWarning(stroke_index, "no_route", "這個部件附近找不到可騎的道路")
        return StrokeMatch(chains=[], warnings=[warning])
    if None in pieces or len(chains) > 1:
        warning = MatchWarning(stroke_index, "partial", "部分路段找不到可騎的道路，已略過")
        return StrokeMatch(chains=chains, warnings=[warning])
    return StrokeMatch(chains=chains, warnings=[])
```

```python
"""串接器 v1（spec §4.4）：把各部件串成一條連續路線。

順序與封閉圈的進出點用直線距離估計（貪婪 + 2-opt），選定後只對實際用到的接點呼叫 route。
"""

from dataclasses import dataclass

from engine.matching.deadline import Deadline
from engine.matching.types import JOIN_TOLERANCE_M, MatchRouter, Segment, segment_from
from engine.routing.geo import haversine_m, polyline_length_m
from engine.routing.router import LonLat, RoutingError

CLOSED_ROTATION_STEP_M = 50.0


@dataclass(frozen=True)
class Part:
    segments: list[Segment]
    closed: bool


@dataclass(frozen=True)
class Composition:
    segments: list[Segment]
    dropped: list[int]


def _flat(segments: list[Segment]) -> list[tuple[int, LonLat]]:
    """(所屬段落索引, 點)；相鄰段共用的接點只算一次。"""
    out: list[tuple[int, LonLat]] = []
    for i, segment in enumerate(segments):
        points = segment.points if not out else segment.points[1:]
        out.extend((i, p) for p in points)
    return out


def _with_points(segment: Segment, points: list[LonLat]) -> Segment:
    return Segment(segment.kind, segment.stroke_index, points, polyline_length_m(points))


def rotate(segments: list[Segment], index: int) -> list[Segment]:
    flat = _flat(segments)
    if index <= 0 or index >= len(flat) - 1:
        return list(segments)
    pivot_seg, pivot = flat[index]
    head: list[Segment] = []
    tail: list[Segment] = []
    for i, segment in enumerate(segments):
        if i < pivot_seg:
            tail.append(segment)
        elif i > pivot_seg:
            head.append(segment)
        else:
            cut = segment.points.index(pivot)
            if cut > 0:
                tail.append(_with_points(segment, segment.points[: cut + 1]))
            if cut < len(segment.points) - 1:
                head.insert(0, _with_points(segment, segment.points[cut:]))
    return head + tail


def _rotation_candidates(part: Part) -> list[int]:
    if not part.closed:
        return [0]
    flat = _flat(part.segments)
    candidates, walked = [0], 0.0
    for k in range(1, len(flat) - 1):
        walked += haversine_m(flat[k - 1][1], flat[k][1])
        if walked >= CLOSED_ROTATION_STEP_M:
            candidates.append(k)
            walked = 0.0
    return candidates


def _entry(part: Part, rotation: int) -> LonLat:
    return _flat(part.segments)[rotation][1]


def _exit(part: Part, rotation: int) -> LonLat:
    return _entry(part, rotation) if part.closed else part.segments[-1].points[-1]


def _plan(parts: list[Part], order: list[int]) -> tuple[float, list[int]]:
    """沿著 order 為每個封閉圈挑最近的進入點；回傳 (總接縫距離, 各 part 的旋轉)。"""
    rotations = [0] * len(parts)
    cost, position = 0.0, None
    for i in order:
        options = _rotation_candidates(parts[i])
        best = options[0] if position is None else min(options, key=lambda r: haversine_m(position, _entry(parts[i], r)))
        rotations[i] = best
        if position is not None:
            cost += haversine_m(position, _entry(parts[i], best))
        position = _exit(parts[i], best)
    return cost, rotations


def _greedy(parts: list[Part]) -> list[int]:
    order, remaining = [0], set(range(1, len(parts)))
    while remaining:
        _, rotations = _plan(parts, order)
        position = _exit(parts[order[-1]], rotations[order[-1]])
        nxt = min(
            remaining,
            key=lambda i: min(haversine_m(position, _entry(parts[i], r)) for r in _rotation_candidates(parts[i])),
        )
        order.append(nxt)
        remaining.remove(nxt)
    return order


def _two_opt(parts: list[Part], order: list[int]) -> list[int]:
    best_cost, _ = _plan(parts, order)
    improved = True
    while improved:
        improved = False
        for i in range(1, len(order) - 1):
            for j in range(i + 1, len(order)):
                candidate = order[:i] + order[i : j + 1][::-1] + order[j + 1 :]
                cost, _ = _plan(parts, candidate)
                if cost < best_cost - 1e-6:
                    order, best_cost, improved = candidate, cost, True
    return order


def compose(router: MatchRouter, parts: list[Part], deadline: Deadline) -> Composition:
    if not parts:
        return Composition(segments=[], dropped=[])
    order = _two_opt(parts, _greedy(parts))
    _, rotations = _plan(parts, order)
    segments: list[Segment] = []
    dropped: list[int] = []
    for i in order:
        body = rotate(parts[i].segments, rotations[i]) if parts[i].closed else parts[i].segments
        start = body[0].points[0]
        if segments and haversine_m(segments[-1].points[-1], start) > JOIN_TOLERANCE_M:
            deadline.check()
            try:
                connector = segment_from("connector", None, router.route(segments[-1].points[-1], start))
            except RoutingError:
                dropped.append(i)
                continue
            segments.append(connector)
        segments.extend(body)
    return Composition(segments=segments, dropped=sorted(dropped))
```

```python
"""貼合度評分（spec §4.5）：在以目標中心為原點的局部平面（公尺）上量距離。"""

import math
from dataclasses import dataclass

import numpy as np

from engine.routing.router import LonLat

SAMPLE_STEP_M = 10.0
ZERO_SCORE_RATIO = 0.05  # 平均偏離達對角線 5% 時為 0 分；由基準測試集調整
_CHUNK = 2048  # 每批計算的取樣點數，限制 N×M 矩陣的記憶體


@dataclass(frozen=True)
class Score:
    coverage: int
    cleanliness: int
    overall: int


def _projector(targets: list[list[LonLat]]):
    lons = [lon for line in targets for lon, _ in line]
    lats = [lat for line in targets for _, lat in line]
    lon0, lat0 = (min(lons) + max(lons)) / 2, (min(lats) + max(lats)) / 2
    kx = 111_320.0 * math.cos(math.radians(lat0))
    ky = 110_574.0

    def project(points: list[LonLat]) -> np.ndarray:
        arr = np.asarray(points, dtype=float).reshape(-1, 2)
        return np.column_stack(((arr[:, 0] - lon0) * kx, (arr[:, 1] - lat0) * ky))

    return project


def _sample(line: np.ndarray) -> np.ndarray:
    parts = [line[:1]]
    for a, b in zip(line, line[1:]):
        n = max(1, math.ceil(float(np.hypot(*(b - a))) / SAMPLE_STEP_M))
        t = np.arange(1, n + 1)[:, None] / n
        parts.append(a + (b - a) * t)
    return np.vstack(parts)


def _segments(lines: list[np.ndarray]) -> tuple[np.ndarray, np.ndarray]:
    starts = np.vstack([line[:-1] for line in lines if len(line) > 1])
    ends = np.vstack([line[1:] for line in lines if len(line) > 1])
    return starts, ends


def _distances(points: np.ndarray, starts: np.ndarray, ends: np.ndarray) -> np.ndarray:
    ab = ends - starts
    ab_len2 = np.maximum((ab**2).sum(axis=1), 1e-12)
    out = np.empty(len(points))
    for i in range(0, len(points), _CHUNK):
        p = points[i : i + _CHUNK, None, :]
        t = np.clip(((p - starts) * ab).sum(axis=2) / ab_len2, 0.0, 1.0)
        nearest = starts + t[..., None] * ab
        out[i : i + _CHUNK] = np.sqrt(((p - nearest) ** 2).sum(axis=2)).min(axis=1)
    return out


def _component(distances: np.ndarray, diagonal: float) -> int:
    e = (distances.mean() + np.percentile(distances, 90)) / 2 / diagonal
    return round(100 * max(0.0, 1 - e / ZERO_SCORE_RATIO))


def score(targets: list[list[LonLat]], route: list[LonLat]) -> Score:
    project = _projector(targets)
    target_lines = [project(line) for line in targets]
    stacked = np.vstack(target_lines)
    diagonal = float(np.hypot(*(stacked.max(axis=0) - stacked.min(axis=0))))
    if diagonal < 1.0:
        raise ValueError("目標圖形太小，無法評分")
    if len(route) < 2:
        return Score(0, 0, 0)
    route_line = project(route)
    target_starts, target_ends = _segments(target_lines)
    route_starts, route_ends = _segments([route_line])
    coverage = _component(
        _distances(np.vstack([_sample(line) for line in target_lines if len(line) > 1]), route_starts, route_ends),
        diagonal,
    )
    cleanliness = _component(_distances(_sample(route_line), target_starts, target_ends), diagonal)
    return Score(coverage, cleanliness, round((coverage + cleanliness) / 2))
```

```python
"""整個圖形的吸附流程：驗證 → 逐部件分治吸附 → 串接 → 評分。"""

from dataclasses import dataclass

from engine.matching.compose import Part, compose
from engine.matching.deadline import Deadline
from engine.matching.score import Score, score
from engine.matching.stroke import match_stroke
from engine.matching.types import JOIN_TOLERANCE_M, MatchRouter, MatchWarning, Segment
from engine.routing.geo import haversine_m, polyline_length_m
from engine.routing.router import LonLat

TAIWAN_BBOX = (118.1, 21.8, 122.1, 26.4)  # 含金門、馬祖
MIN_TARGET_LENGTH_M = 50.0
RADIUS_RATIO, MIN_RADIUS_M, MAX_RADIUS_M = 0.02, 30.0, 150.0


class OutsideTaiwan(Exception):
    pass


class TargetTooSmall(Exception):
    pass


class AllStrokesFailed(Exception):
    def __init__(self, warnings: list[MatchWarning]) -> None:
        super().__init__("所有部件都找不到可騎的道路")
        self.warnings = warnings


@dataclass(frozen=True)
class StrokeInput:
    points: list[LonLat]
    closed: bool


@dataclass(frozen=True)
class MatchResult:
    route_points: list[LonLat]
    route_length_m: float
    segments: list[Segment]
    score: Score
    target_length_m: float
    warnings: list[MatchWarning]


def _target(stroke: StrokeInput) -> list[LonLat]:
    return stroke.points + [stroke.points[0]] if stroke.closed else stroke.points


def _diagonal_m(strokes: list[StrokeInput]) -> float:
    lons = [lon for s in strokes for lon, _ in s.points]
    lats = [lat for s in strokes for _, lat in s.points]
    return haversine_m((min(lons), min(lats)), (max(lons), max(lats)))


def default_search_radius(strokes: list[StrokeInput]) -> float:
    return min(MAX_RADIUS_M, max(MIN_RADIUS_M, _diagonal_m(strokes) * RADIUS_RATIO))


def _check_inside_taiwan(strokes: list[StrokeInput]) -> None:
    min_lon, min_lat, max_lon, max_lat = TAIWAN_BBOX
    for stroke in strokes:
        for lon, lat in stroke.points:
            if not (min_lon <= lon <= max_lon and min_lat <= lat <= max_lat):
                raise OutsideTaiwan("圖形不在台灣範圍內")


def _closes(chain: list[Segment]) -> bool:
    return haversine_m(chain[-1].points[-1], chain[0].points[0]) <= JOIN_TOLERANCE_M


def _route_points(segments: list[Segment]) -> list[LonLat]:
    points: list[LonLat] = []
    for segment in segments:
        points.extend(segment.points if not points else segment.points[1:])
    return points


def match_shape(
    router: MatchRouter,
    strokes: list[StrokeInput],
    *,
    search_radius_m: float | None,
    deadline: Deadline,
) -> MatchResult:
    _check_inside_taiwan(strokes)
    targets = [_target(s) for s in strokes]
    target_length = sum(polyline_length_m(t) for t in targets)
    if target_length < MIN_TARGET_LENGTH_M:
        raise TargetTooSmall(f"圖形總長不到 {MIN_TARGET_LENGTH_M:.0f} m")
    radius = search_radius_m or default_search_radius(strokes)

    parts: list[Part] = []
    warnings: list[MatchWarning] = []
    for index, target in enumerate(targets):
        matched = match_stroke(
            router, target, stroke_index=index, search_radius_m=radius, min_piece_m=2 * radius, deadline=deadline
        )
        warnings.extend(matched.warnings)
        closed = strokes[index].closed and len(matched.chains) == 1 and _closes(matched.chains[0])
        parts.extend(Part(chain, closed) for chain in matched.chains)
    if not parts:
        raise AllStrokesFailed(warnings)

    composition = compose(router, parts, deadline)
    for i in composition.dropped:
        stroke_index = parts[i].segments[0].stroke_index
        warnings.append(MatchWarning(stroke_index, "partial", "這段無法與其他部分相連，已略過"))
    route = _route_points(composition.segments)
    return MatchResult(
        route_points=route,
        route_length_m=sum(s.length_m for s in composition.segments),
        segments=composition.segments,
        score=score(targets, route),
        target_length_m=target_length,
        warnings=warnings,
    )
```

```python
LonLatPair = tuple[float, float]


class StrokeIn(BaseModel):

    strokes: list[StrokeIn] = Field(min_length=1, max_length=40)
    search_radius_m: float | None = Field(default=None, ge=30, le=150)


class RouteOut(BaseModel):
    points: list[LonLatPair]
    length_m: float


class SegmentOut(BaseModel):
    kind: Literal["stroke", "detour", "connector"]
    stroke_index: int | None
    points: list[LonLatPair]
    length_m: float


class ScoreOut(BaseModel):
    coverage: int
    cleanliness: int
    overall: int


class WarningOut(BaseModel):
    stroke_index: int
    code: Literal["no_route", "partial"]
    message: str


class MatchResponse(BaseModel):
    route: RouteOut
    segments: list[SegmentOut]
    score: ScoreOut
    target_length_m: float
    warnings: list[WarningOut]
    data_version: str


class BuildInfo(BaseModel):

    osm_md5: str
    created_at: str
```

```python
"""HTTP 介面。只在 hl 內網提供服務，由 web 轉發請求。"""

import os
from collections.abc import Callable
from dataclasses import asdict
from pathlib import Path
from typing import Any

from fastapi import FastAPI, HTTPException

from engine.api.models import BuildInfo, MatchRequest, MatchResponse, StatusResponse
from engine.data.layout import DataLayout
from engine.data.pipeline import read_meta
from engine.matching.deadline import Deadline, MatchTimeout
from engine.matching.service import AllStrokesFailed, MatchResult, OutsideTaiwan, StrokeInput, TargetTooSmall, match_shape
from engine.routing.router import Router
from engine.service.current import CurrentRouter, NoDataError


def _error(status: int, code: str, message: str, **extra: Any) -> HTTPException:
    return HTTPException(status_code=status, detail={"code": code, "message": message, **extra})


def _response(result: MatchResult, version: str) -> MatchResponse:
    return MatchResponse.model_validate(
        {
            "route": {"points": result.route_points, "length_m": result.route_length_m},
            "segments": [asdict(s) for s in result.segments],
            "score": asdict(result.score),
            "target_length_m": result.target_length_m,
            "warnings": [asdict(w) for w in result.warnings],
            "data_version": version,
        }
    )


def create_app(
    data_root: Path,
    *,
    router_factory: Callable[[Path], Any] = Router.from_build,
    match_timeout_s: float = 5.0,
) -> FastAPI:
    app = FastAPI(title="artroute-engine")
    layout = DataLayout(data_root)
    current = CurrentRouter(layout, router_factory)

    @app.get("/")
    def root() -> dict[str, str]:
        return {"service": "artroute-engine"}

    @app.get("/health")
    def health() -> dict[str, bool]:
        return {"ok": True}

    @app.get("/status")
    def status() -> StatusResponse:
        build = layout.current()
        meta = read_meta(build) if build else None
        return StatusResponse(data=BuildInfo(**asdict(meta)) if meta else None)

    @app.post("/match")
    def match(request: MatchRequest) -> MatchResponse:
        deadline = Deadline(match_timeout_s)  # 從進入請求開始算，含等 Router 鎖的時間
        try:
            router, version = current.get()
        except NoDataError as error:
            raise _error(503, "no_data", str(error)) from error
        strokes = [StrokeInput(points=list(s.points), closed=s.closed) for s in request.strokes]
        try:
            result = match_shape(router, strokes, search_radius_m=request.search_radius_m, deadline=deadline)
        except OutsideTaiwan as error:
            raise _error(400, "outside_taiwan", str(error)) from error
        except TargetTooSmall as error:
            raise _error(422, "too_small", str(error)) from error
        except AllStrokesFailed as error:
            raise _error(422, "no_route", str(error), warnings=[asdict(w) for w in error.warnings]) from error
        except MatchTimeout as error:
            raise _error(504, "timeout", "吸附超過時間上限，請縮小圖形或減少部件") from error
        return _response(result, version)

    return app


app = create_app(Path(os.environ.get("DATA_DIR", "/data")))
```

```python
"""建圖前把台灣快速道路標成 bicycle=no（spec §6.1）。

台灣的快速道路多半標成 highway=trunk，Valhalla 對 bicycle 視為可騎；
但一般省道（台1線等）也是 trunk，所以用名稱、編號與 motorroad 判斷，而不是全面禁止 trunk。
"""

import tomllib
from collections.abc import Mapping
from dataclasses import dataclass
from pathlib import Path

import osmium

EXPRESSWAY_REFS = frozenset(
    {"台61", "台61甲", "台62", "台62甲", "台64", "台65", "台66", "台68",
     "台72", "台74", "台76", "台78", "台82", "台84", "台86", "台88"}
)
NAME_KEYWORDS = ("快速道路", "快速公路", "快速路", "高架")
TRUNK = frozenset({"trunk", "trunk_link"})


def is_expressway(tags: Mapping[str, str]) -> bool:
    if tags.get("highway") not in TRUNK:
        return False
    if tags.get("motorroad") == "yes":
        return True
    if any(keyword in tags.get("name", "") for keyword in NAME_KEYWORDS):
        return True
    refs = {ref.strip() for ref in tags.get("ref", "").split(";")}
    return bool(refs & EXPRESSWAY_REFS)


@dataclass(frozen=True)
class Overrides:
    deny: frozenset[int]
    allow: frozenset[int]


def load_overrides(path: Path) -> Overrides:
    data = tomllib.loads(path.read_text())
    return Overrides(deny=frozenset(data.get("deny", [])), allow=frozenset(data.get("allow", [])))


@dataclass(frozen=True)
class DeniedWay:
    id: int
    name: str
    ref: str
    first: tuple[float, float]
    last: tuple[float, float]


def _tags(obj) -> dict[str, str]:
    return {tag.k: tag.v for tag in obj.tags}


def _denied(way_id: int, tags: Mapping[str, str], overrides: Overrides) -> bool:
    return (is_expressway(tags) or way_id in overrides.deny) and way_id not in overrides.allow


def patch_pbf(src: Path, dst: Path, overrides: Overrides) -> list[DeniedWay]:
    # 第一遍只看 way：找出要標記的 way 與它們首尾節點的 id
    targets: dict[int, tuple[dict[str, str], int, int]] = {}
    for way in osmium.FileProcessor(str(src), osmium.osm.WAY):
        tags = _tags(way)
        if _denied(way.id, tags, overrides):
            targets[way.id] = (tags, way.nodes[0].ref, way.nodes[-1].ref)
    needed = {ref for _, first, last in targets.values() for ref in (first, last)}

    # 第二遍完整複製；節點在 way 之前，順便記下需要的座標
    locations: dict[int, tuple[float, float]] = {}
    writer = osmium.SimpleWriter(str(dst), overwrite=True)
    try:
        for obj in osmium.FileProcessor(str(src)):
            if obj.is_node():
                if obj.id in needed:
                    locations[obj.id] = (obj.location.lon, obj.location.lat)
                writer.add_node(obj)
            elif obj.is_way():
                if obj.id in targets:
                    writer.add_way(obj.replace(tags={**targets[obj.id][0], "bicycle": "no"}))
                else:
                    writer.add_way(obj)
            elif obj.is_relation():
                writer.add_relation(obj)
    finally:
        writer.close()

    return [
        DeniedWay(way_id, tags.get("name", ""), tags.get("ref", ""), locations[first], locations[last])
        for way_id, (tags, first, last) in sorted(targets.items())
    ]
```

```python
    finally:
        patched.unlink(missing_ok=True)

    raw = json.loads((build_dir / DENIED_FILE).read_text())
    return [DeniedWay(d["id"], d["name"], d["ref"], tuple(d["first"]), tuple(d["last"])) for d in raw]
```

```python
"""台灣真實路網的驗證查詢：新版路網必須全部通過才會啟用。"""

from pathlib import Path
from typing import Protocol

from engine.data.bike_patch import DeniedWay
from engine.data.build import load_denied
from engine.routing.router import EdgeInfo, LonLat, RouteResult, Router, RoutingError

# 洲美街單行段（OSM way 287016294，oneway=yes，節點順序由西往東）。座標取自該 way 的 geometry
ZHOUMEI_ONEWAY_WAY_ID = 287016294
ZHOUMEI_WEST: LonLat = (121.4994400, 25.1010677)
ZHOUMEI_EAST: LonLat = (121.5035557, 25.1011282)
ZHOUMEI_EXPRESSWAY_NAME = "洲美快速道路"


class RouterLike(Protocol):
    def route(self, start: LonLat, end: LonLat) -> RouteResult: ...
    def edges(self, points: list[LonLat]) -> list[EdgeInfo]: ...


def _oneway_checks(router: RouterLike) -> list[str]:
    failures: list[str] = []
    forward = router.route(ZHOUMEI_WEST, ZHOUMEI_EAST)
    if not any(e.way_id == ZHOUMEI_ONEWAY_WAY_ID for e in router.edges(forward.points)):
        failures.append(
            f"順向路線沒有走洲美街單行段（way {ZHOUMEI_ONEWAY_WAY_ID}）：OSM 資料可能變了，需要更新這項檢查"
        )
    backward = router.route(ZHOUMEI_EAST, ZHOUMEI_WEST)
    if backward.length_km <= forward.length_km:
        failures.append("洲美街逆向路線沒有比順向長：可能沒有真的繞路")
    for edge in router.edges(backward.points):
        if edge.way_id == ZHOUMEI_ONEWAY_WAY_ID and not 45 <= edge.begin_heading <= 135:
            failures.append(f"逆向騎上洲美街單行段（heading {edge.begin_heading}）")
    return failures


def _expressway_checks(router: RouterLike, denied: list[DeniedWay]) -> list[str]:
    if not denied:
        return ["沒有任何快速道路被標記為禁行：資料或標記規則可能壞了"]
    zhoumei = [d for d in denied if d.name == ZHOUMEI_EXPRESSWAY_NAME]
    if not zhoumei:
        return [f"找不到{ZHOUMEI_EXPRESSWAY_NAME}：OSM 資料可能變了，需要更新這項檢查"]
    denied_ids = {d.id for d in denied}
    route = router.route(zhoumei[0].first, zhoumei[0].last)
    used = {e.way_id for e in router.edges(route.points)} & denied_ids
    if used:
        return [f"自行車路線使用了被禁行的快速道路 way {sorted(used)}"]
    return []


def taiwan_checks(router: RouterLike, denied: list[DeniedWay]) -> list[str]:
    failures: list[str] = []
    for name, check in (("洲美街單行段檢查", lambda: _oneway_checks(router)),
                        ("洲美快速道路檢查", lambda: _expressway_checks(router, denied))):
        try:
            failures.extend(check())
        except RoutingError as error:
            failures.append(f"{name}失敗：{error}")
    return failures


def validate_build(build_dir: Path) -> list[str]:
    return taiwan_checks(Router.from_build(build_dir), load_denied(build_dir))
```

```python
"""基準測試案例：幾個代表性圖形 × 台灣幾種街道型態的地點。"""

import math
from dataclasses import dataclass

from engine.matching.service import StrokeInput

type Point = tuple[float, float]


@dataclass(frozen=True)
class Shape:
    name: str
    strokes: list[tuple[list[Point], bool]]


@dataclass(frozen=True)
class Case:
    name: str
    shape: Shape
    center: Point
    size_m: float
    rotation_deg: float


def _heart(n: int = 48) -> list[Point]:
    raw = [
        (16 * math.sin(t) ** 3, 13 * math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t))
        for t in (2 * math.pi * i / n for i in range(n))
    ]
    xs, ys = [p[0] for p in raw], [p[1] for p in raw]
    return [((x - min(xs)) / (max(xs) - min(xs)), (y - min(ys)) / (max(ys) - min(ys))) for x, y in raw]


def _star() -> list[Point]:
    pts = []
    for i in range(10):
        r = 0.5 if i % 2 == 0 else 0.2
        a = math.pi / 2 + i * math.pi / 5
        pts.append((0.5 + r * math.cos(a), 0.5 + r * math.sin(a)))
    return pts


SHAPES: list[Shape] = [
    Shape("heart", [(_heart(), True)]),
    Shape("star", [(_star(), True)]),
    Shape("letter_l", [([(0.2, 1.0), (0.2, 0.0), (0.8, 0.0)], False)]),
    Shape("square", [([(0, 0), (1, 0), (1, 1), (0, 1)], True)]),
]

    "taipei_daan": (121.5435, 25.0330),        # 棋盤狀街廓、單行道多
    "taichung_west": (120.6620, 24.1470),      # 放射＋棋盤

def place(shape: Shape, center: Point, size_m: float, rotation_deg: float) -> list[StrokeInput]:
    """把正規化圖形放到地圖上：以中心為原點、邊長 size_m，逆時針旋轉 rotation_deg。"""
    lon0, lat0 = center
    kx = 111_320.0 * math.cos(math.radians(lat0))
    ky = 110_574.0
    cos_r, sin_r = math.cos(math.radians(rotation_deg)), math.sin(math.radians(rotation_deg))

    def to_lonlat(p: Point) -> Point:
        x, y = (p[0] - 0.5) * size_m, (p[1] - 0.5) * size_m
        x, y = x * cos_r - y * sin_r, x * sin_r + y * cos_r
        return (lon0 + x / kx, lat0 + y / ky)

    return [StrokeInput([to_lonlat(p) for p in points], closed) for points, closed in shape.strokes]
```

```python
"""執行基準測試並與 baseline 比較。"""

import time
from dataclasses import dataclass

from engine.bench.cases import Case, place
from engine.matching.deadline import Deadline, MatchTimeout
from engine.matching.service import AllStrokesFailed, match_shape


@dataclass(frozen=True)
class CaseResult:
    name: str
    coverage: int
    cleanliness: int
    overall: int
    length_m: float
    target_length_m: float
    ms: int
    error: str | None


def _run_one(router, case: Case, timeout_s: float) -> CaseResult:
    strokes = place(case.shape, case.center, case.size_m, case.rotation_deg)
    started = time.perf_counter()
    try:
        result = match_shape(router, strokes, search_radius_m=None, deadline=Deadline(timeout_s))
    except (AllStrokesFailed, MatchTimeout) as error:
        ms = round((time.perf_counter() - started) * 1000)
        return CaseResult(case.name, 0, 0, 0, 0.0, 0.0, ms, type(error).__name__)
    ms = round((time.perf_counter() - started) * 1000)
    s = result.score
    return CaseResult(
        case.name, s.coverage, s.cleanliness, s.overall,
        round(result.route_length_m, 1), round(result.target_length_m, 1), ms, None,
    )


def run_cases(router, cases: list[Case], timeout_s: float = 30.0) -> list[CaseResult]:
    return [_run_one(router, case, timeout_s) for case in cases]


def compare(results: list[CaseResult], baseline: dict[str, dict], tolerance: int = 3) -> list[str]:
    problems: list[str] = []
    for r in results:
        base = baseline.get(r.name)
        if base is None:
            continue
        if r.error and not base.get("error"):
            problems.append(f"{r.name}：原本成功，現在失敗（{r.error}）")
        elif r.overall < base["overall"] - tolerance:
            problems.append(f"{r.name}：overall {base['overall']} → {r.overall}")
    return problems
```
